CLASS lcl_currency_converter DEFINITION FINAL.
  PUBLIC SECTION.
    "! Currency converter (SRP): converts an amount between currencies using standard API (no DB access).
    CLASS-METHODS convert_amount
      IMPORTING
        "! Conversion parameters
        is_params        TYPE ty_conversion_params
      RETURNING
        VALUE(rs_result) TYPE ty_conversion_result.
ENDCLASS.

CLASS lcl_currency_converter IMPLEMENTATION.
  METHOD convert_amount.
    rs_result = VALUE ty_conversion_result(
      amount_to = is_params-amount_from
      status    = 'E'
      message   = `` ).

    IF is_params-currency_from IS INITIAL OR is_params-currency_to IS INITIAL.
      rs_result-message = |Missing currency (from/to).|.
      RETURN.
    ENDIF.

    IF is_params-currency_from = is_params-currency_to.
      rs_result-status  = 'S'.
      rs_result-message = ``.
      RETURN.
    ENDIF.

    DATA(lv_exch_type) = COND kurst( WHEN is_params-exch_type IS INITIAL THEN 'M'
                                    ELSE is_params-exch_type ).

    TRY.
        DATA(lv_amount_to) = VALUE wrbtr( ).

        CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
          EXPORTING
            date             = is_params-conv_date
            foreign_amount   = is_params-amount_from
            foreign_currency = is_params-currency_from
            local_currency   = is_params-currency_to
            type_of_rate     = lv_exch_type
          IMPORTING
            local_amount     = lv_amount_to
          EXCEPTIONS
            no_rate_found    = 1
            overflow         = 2
            no_factors_found = 3
            no_spread_found  = 4
            derived_2_times  = 5
            OTHERS           = 6.

        IF sy-subrc = 0.
          rs_result-amount_to = lv_amount_to.
          rs_result-status    = 'S'.
          rs_result-message   = ``.
          RETURN.
        ENDIF.

        rs_result-amount_to = is_params-amount_from.
        rs_result-status    = 'E'.
        rs_result-message   = |Conversion failed (subrc={ sy-subrc }).|.
        RETURN.

      CATCH cx_sy_dyn_call_illegal_func INTO DATA(lx_no_fm).
        rs_result-amount_to = is_params-amount_from.
        rs_result-status    = 'E'.
        rs_result-message   = |Conversion FM not available: { lx_no_fm->get_text( ) }|.
        RETURN.

      CATCH cx_sy_conversion_error INTO DATA(lx_conv).
        rs_result-amount_to = is_params-amount_from.
        rs_result-status    = 'E'.
        rs_result-message   = |Conversion error: { lx_conv->get_text( ) }|.
        RETURN.

      CATCH cx_sy_arithmetic_error INTO DATA(lx_arith).
        rs_result-amount_to = is_params-amount_from.
        rs_result-status    = 'E'.
        rs_result-message   = |Arithmetic error: { lx_arith->get_text( ) }|.
        RETURN.

      CATCH cx_root INTO DATA(lx_root).
        rs_result-amount_to = is_params-amount_from.
        rs_result-status    = 'E'.
        rs_result-message   = |Unexpected error: { lx_root->get_text( ) }|.
        RETURN.
    ENDTRY.
  ENDMETHOD.
ENDCLASS.

CLASS lcl_aging_service DEFINITION FINAL.
  PUBLIC SECTION.
    "! Service (SRP): enriches open items with aging logic + currency conversion + traffic-light indicators.
    "! No DB access (no SELECT); pure business enrichment.
    CLASS-METHODS build_output
      IMPORTING
        "! Selection/configuration (reporting currency, exchange rate type, company code)
        is_sel   TYPE ty_selection
        "! Source open items (AR/AP) without business logic applied
        it_items TYPE tt_item_src
      RETURNING
        "! Enriched output for ALV
        VALUE(rt_out) TYPE tt_out.
ENDCLASS.

CLASS lcl_aging_service IMPLEMENTATION.
  METHOD build_output.
    DATA(lv_rep_waers) = COND waers( WHEN is_sel-rep_waers IS INITIAL THEN 'USD'
                                    ELSE is_sel-rep_waers ).

    rt_out = VALUE #( ).

    LOOP AT it_items ASSIGNING FIELD-SYMBOL(<ls_item>).
      DATA(ls_out) = CORRESPONDING ty_out( <ls_item> ).

      ls_out-rep_waers = lv_rep_waers.

      " Days open + bucket/status/text/icon
      IF <ls_item>-budat IS INITIAL.
        ls_out-days_open   = 0.
        ls_out-bucket      = '0-30'.
        ls_out-bucket_text = '0-30 days'.
        ls_out-status      = gc_status_yellow.
        ls_out-status_text = 'Warning'.
        ls_out-icon        = icon_yellow_light.

        ls_out-conv_msg = COND string(
          WHEN ls_out-conv_msg IS INITIAL THEN |Missing posting date (BUDAT).|
          ELSE ls_out-conv_msg ).
      ELSE.
        ls_out-days_open = sy-datum - <ls_item>-budat.

        IF ls_out-days_open <= 30.
          ls_out-bucket      = '0-30'.
          ls_out-bucket_text = '0-30 days'.
          ls_out-status      = gc_status_green.
          ls_out-status_text = 'Green'.
          ls_out-icon        = icon_green_light.
        ELSEIF ls_out-days_open <= 60.
          ls_out-bucket      = '31-60'.
          ls_out-bucket_text = '31-60 days'.
          ls_out-status      = gc_status_yellow.
          ls_out-status_text = 'Yellow'.
          ls_out-icon        = icon_yellow_light.
        ELSEIF ls_out-days_open <= 90.
          ls_out-bucket      = '61-90'.
          ls_out-bucket_text = '61-90 days'.
          ls_out-status      = gc_status_orange.
          ls_out-status_text = 'Orange'.
          ls_out-icon        = icon_yellow_light.
        ELSE.
          ls_out-bucket      = '>90'.
          ls_out-bucket_text = '>90 days'.
          ls_out-status      = gc_status_red.
          ls_out-status_text = 'Red'.
          ls_out-icon        = icon_red_light.
        ENDIF.
      ENDIF.

      " Currency conversion (delegates to lcl_currency_converter)
      IF <ls_item>-waers IS INITIAL.
        ls_out-amount_rep = <ls_item>-amount_orig.
        ls_out-conv_error = abap_true.
        ls_out-conv_msg   = COND string(
          WHEN ls_out-conv_msg IS INITIAL THEN |Missing source currency (WAERS).|
          ELSE ls_out-conv_msg && ` ` && |Missing source currency (WAERS).| ).
      ELSEIF <ls_item>-waers = lv_rep_waers.
        ls_out-amount_rep = <ls_item>-amount_orig.
        ls_out-conv_error = abap_false.
      ELSE.
        DATA(ls_conv) = lcl_currency_converter=>convert_amount(
          VALUE ty_conversion_params(
            amount_from   = <ls_item>-amount_orig
            currency_from = <ls_item>-waers
            currency_to   = lv_rep_waers
            conv_date     = <ls_item>-budat
            exch_type     = is_sel-ex_rate_type
            company_code  = is_sel-bukrs ) ).

        ls_out-amount_rep = ls_conv-amount_to.
        ls_out-conv_error = COND abap_bool( WHEN ls_conv-status = 'E' THEN abap_true ELSE abap_false ).
        ls_out-conv_msg   = ls_conv-message.
      ENDIF.

      APPEND ls_out TO rt_out.
    ENDLOOP.
  ENDMETHOD.
ENDCLASS.
